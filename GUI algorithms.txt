import tkinter as tk
import time, threading, random

# ---------------- COLORS ----------------
COLORS = {
    'U': 'white',
    'D': 'yellow',
    'L': 'blue',
    'R': 'green',
    'F': 'red',
    'B': 'orange'
}

# Flattened cube order
FACES = ['U','R','F','D','L','B']

SOLVED = (
    ['U']*9 +
    ['R']*9 +
    ['F']*9 +
    ['D']*9 +
    ['L']*9 +
    ['B']*9
)

cube = SOLVED[:]
scramble_moves = []

# ---------------- ROTATION HELPERS ----------------
def rotate_face(face_idx):
    f = cube[face_idx:face_idx+9]
    cube[face_idx:face_idx+9] = [
        f[6], f[3], f[0],
        f[7], f[4], f[1],
        f[8], f[5], f[2]
    ]

# ---------------- TRUE MOVES ----------------
def move_U():
    rotate_face(0)
    a,b,c = cube[18:21], cube[9:12], cube[45:48]
    cube[18:21] = cube[36:39]
    cube[36:39] = c
    cube[45:48] = b
    cube[9:12]  = a

def move_R():
    rotate_face(9)
    a = [cube[i] for i in [2,5,8]]
    b = [cube[i] for i in [20,23,26]]
    c = [cube[i] for i in [29,32,35]]
    d = [cube[i] for i in [47,50,53]]

    for i,v in zip([2,5,8], b): cube[i]=v
    for i,v in zip([20,23,26], c): cube[i]=v
    for i,v in zip([29,32,35], d[::-1]): cube[i]=v
    for i,v in zip([47,50,53], a[::-1]): cube[i]=v

def move_F():
    rotate_face(18)
    a = [cube[i] for i in [6,7,8]]
    b = [cube[i] for i in [36,39,42]]
    c = [cube[i] for i in [27,28,29]]
    d = [cube[i] for i in [11,14,17]]

    for i,v in zip([6,7,8], b[::-1]): cube[i]=v
    for i,v in zip([36,39,42], c): cube[i]=v
    for i,v in zip([27,28,29], d[::-1]): cube[i]=v
    for i,v in zip([11,14,17], a): cube[i]=v

MOVES = {
    'U': move_U,
    'R': move_R,
    'F': move_F
}

# ---------------- GUI ----------------
root = tk.Tk()
root.title("Rubik's Cube – TRUE PHYSICS")

canvas = tk.Canvas(root, width=500, height=350, bg="#eef")
canvas.pack()

status = tk.Label(root, text="Ready", font=("Arial",12))
status.pack()

def draw():
    canvas.delete("all")
    face_pos = {
        'U':(170,20),'L':(70,110),'F':(170,110),
        'R':(270,110),'B':(370,110),'D':(170,200)
    }
    for f, (x,y) in face_pos.items():
        idx = FACES.index(f)*9
        for i in range(9):
            r,c = divmod(i,3)
            canvas.create_rectangle(
                x+c*30,y+r*30,x+c*30+28,y+r*30+28,
                fill=COLORS[cube[idx+i]],
                outline="black"
            )

def scramble():
    global cube, scramble_moves
    cube = SOLVED[:]
    scramble_moves = []
    for _ in range(15):
        m = random.choice(list(MOVES.keys()))
        scramble_moves.append(m)
        MOVES[m]()
    draw()
    status.config(text="Scrambled")

def solve():
    def run():
        status.config(text="Solving...")
        for m in reversed(scramble_moves):
            for _ in range(3):  # inverse
                MOVES[m]()
            draw()
            time.sleep(0.25)
        status.config(text="Solved ✔")
    threading.Thread(target=run, daemon=True).start()

tk.Button(root,text="SCRAMBLE",command=scramble,width=12).pack(side="left",padx=10)
tk.Button(root,text="SOLVE",command=solve,width=12,bg="green",fg="white").pack(side="right",padx=10)

draw()
root.mainloop()
